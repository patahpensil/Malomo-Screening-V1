//@version=6
indicator("Claude Trading Indicator", shorttitle="CTI", overlay=true, max_labels_count=500)

// =====================================================================
// CLAUDE TRADING INDICATOR
// Votes are grouped into 4 market dimensions: Trend, Momentum,
// Volatility, and Volume. Each dimension aggregates its own component
// indicators internally (Bullish only if ALL its components agree,
// Bearish only if ALL agree, otherwise Neutral). A final BUY/SELL signal
// only fires when every currently enabled dimension is aligned in the
// same direction — a two-tier unanimous consensus system.
// =====================================================================

// ---------------------- COLOR PALETTE ----------------------
colBull      = color.new(#1FA37A, 0)
colBear      = color.new(#E5484D, 0)
colBullSoft  = color.new(#1FA37A, 88)
colBearSoft  = color.new(#E5484D, 88)
colNeutral   = color.new(#787B86, 40)
colPanelBg   = color.new(#131722, 15)
colPanelText = color.new(#D1D4DC, 0)

// ============================ INPUTS ============================

grpCat = "Market Dimensions"
useTrend      = input.bool(true, "Enable Trend", group=grpCat, tooltip="Includes this dimension in the final verdict.")
useMomentum   = input.bool(true, "Enable Momentum", group=grpCat, tooltip="Includes this dimension in the final verdict.")
useVolatility = input.bool(true, "Enable Volatility", group=grpCat, tooltip="Includes this dimension in the final verdict.")
useVolume     = input.bool(true, "Enable Volume", group=grpCat, tooltip="Includes this dimension in the final verdict.")

// Internal defaults for each component indicator. Not exposed in the
// inputs menu — only the 4 dimension toggles, risk management, and
// signal filtering settings are user-adjustable.
emaLen      = 200
stAtrPeriod = 10
stFactor    = 3.0
bbLen       = 20
bbMult      = 2.0

macdFast = 12
macdSlow = 26
macdSig  = 9
rsiLen   = 14
stochK      = 14
stochD      = 3
stochSmooth = 3

adxLen         = 14
adxSmooth      = 14
adxMinStrength = 20.0

volSmoothLen = 10

grpRisk = "Risk Management"
atrLen  = input.int(14, "ATR Length (Stop Loss)", minval=1, group=grpRisk)
atrMult = input.float(3.0, "ATR Multiplier (Stop Loss)", minval=0.1, step=0.1, group=grpRisk)

grpVis = "Visual"
showDashboard  = input.bool(true, "Show Dashboard Table", group=grpVis)
showBackground = input.bool(true, "Show Verdict Background Tint", group=grpVis)

grpFilter = "Signal Filtering"
persistBars    = input.int(2, "Require Verdict For (bars)", minval=1, group=grpFilter, tooltip="The final verdict must hold for this many consecutive bars before a signal fires.")
minBarsBetween = input.int(10, "Minimum Bars Between Signals", minval=0, group=grpFilter, tooltip="Blocks new signals for this many bars after the last one.")

// ============================ TREND DIMENSION ============================
emaVal = ta.ema(close, emaLen)
emaVote = close > emaVal ? 1 : close < emaVal ? -1 : 0

[stLine, stDir] = ta.supertrend(stFactor, stAtrPeriod)
stVote = stDir < 0 ? 1 : stDir > 0 ? -1 : 0

bbBasis = ta.sma(close, bbLen)
bbDev   = bbMult * ta.stdev(close, bbLen)
bbUpper = bbBasis + bbDev
bbLower = bbBasis - bbDev
bbVote  = close > bbBasis ? 1 : close < bbBasis ? -1 : 0

trendVote = (emaVote == 1 and stVote == 1 and bbVote == 1) ? 1 :
     (emaVote == -1 and stVote == -1 and bbVote == -1) ? -1 : 0

// ============================ MOMENTUM DIMENSION ============================
[macdLine, macdSignalLine, _] = ta.macd(close, macdFast, macdSlow, macdSig)
macdVote = macdLine > macdSignalLine ? 1 : macdLine < macdSignalLine ? -1 : 0

rsiVal  = ta.rsi(close, rsiLen)
rsiVote = rsiVal > 50 ? 1 : rsiVal < 50 ? -1 : 0

stochRaw  = ta.stoch(close, high, low, stochK)
kLine     = ta.sma(stochRaw, stochSmooth)
dLine     = ta.sma(kLine, stochD)
stochVote = kLine > dLine ? 1 : kLine < dLine ? -1 : 0

momentumVote = (macdVote == 1 and rsiVote == 1 and stochVote == 1) ? 1 :
     (macdVote == -1 and rsiVote == -1 and stochVote == -1) ? -1 : 0

// ============================ VOLATILITY DIMENSION ============================
[diPlus, diMinus, adxValue] = ta.dmi(adxLen, adxSmooth)
adxTrending = adxValue >= adxMinStrength
volatilityVote = adxTrending ? (diPlus > diMinus ? 1 : diPlus < diMinus ? -1 : 0) : 0

// ============================ VOLUME DIMENSION ============================
obvVal = ta.obv
obvMa  = ta.sma(obvVal, volSmoothLen)
volumeVote = obvVal > obvMa ? 1 : obvVal < obvMa ? -1 : 0

// ============================ ATR FOR STOP LOSS ============================
atrVal = ta.atr(atrLen)

// ============================ FINAL VERDICT LOGIC ============================
enabledCount = (useTrend ? 1 : 0) + (useMomentum ? 1 : 0) + (useVolatility ? 1 : 0) + (useVolume ? 1 : 0)

bullCount = (useTrend and trendVote == 1 ? 1 : 0) + (useMomentum and momentumVote == 1 ? 1 : 0) +
     (useVolatility and volatilityVote == 1 ? 1 : 0) + (useVolume and volumeVote == 1 ? 1 : 0)

bearCount = (useTrend and trendVote == -1 ? 1 : 0) + (useMomentum and momentumVote == -1 ? 1 : 0) +
     (useVolatility and volatilityVote == -1 ? 1 : 0) + (useVolume and volumeVote == -1 ? 1 : 0)

verdictBull = enabledCount > 0 and bullCount == enabledCount
verdictBear = enabledCount > 0 and bearCount == enabledCount

var int bullStreak = 0
var int bearStreak = 0
bullStreak := verdictBull ? bullStreak + 1 : 0
bearStreak := verdictBear ? bearStreak + 1 : 0

bullReady = bullStreak == persistBars
bearReady = bearStreak == persistBars

var int lastSignalBar = na
cooldownOk = na(lastSignalBar) or (bar_index - lastSignalBar) >= minBarsBetween

buySignal  = bullReady and cooldownOk
sellSignal = bearReady and cooldownOk and not buySignal

if buySignal or sellSignal
    lastSignalBar := bar_index

longStop  = close - atrVal * atrMult
shortStop = close + atrVal * atrMult

// Track the active stop level for the duration of the trade, so it draws
// as a visible line on the chart instead of a single, easy-to-miss dot.
var float activeLongStop  = na
var float activeShortStop = na
var bool  inLongTrade  = false
var bool  inShortTrade = false

if buySignal
    inLongTrade    := true
    inShortTrade   := false
    activeLongStop := longStop

if sellSignal
    inShortTrade    := true
    inLongTrade     := false
    activeShortStop := shortStop

// Stop the line once price actually trades through the stop level
if inLongTrade and low <= activeLongStop
    inLongTrade := false
if inShortTrade and high >= activeShortStop
    inShortTrade := false

// ============================ VISUALS ============================
plot(useTrend ? emaVal : na, "EMA Trend", color=color.new(#5B8DEF, 0), linewidth=2)
bbTop = plot(useTrend ? bbUpper : na, "BB Upper", color=colNeutral)
bbBot = plot(useTrend ? bbLower : na, "BB Lower", color=colNeutral)
fill(bbTop, bbBot, color=useTrend ? color.new(#5B8DEF, 95) : na)

plotshape(buySignal, title="BUY", style=shape.triangleup, location=location.belowbar, color=colBull, size=size.small)
plotshape(sellSignal, title="SELL", style=shape.triangledown, location=location.abovebar, color=colBear, size=size.small)

plot(inLongTrade ? activeLongStop : na, "Long Stop Loss", style=plot.style_linebr, color=colBull, linewidth=2)
plot(inShortTrade ? activeShortStop : na, "Short Stop Loss", style=plot.style_linebr, color=colBear, linewidth=2)

bgcolor(showBackground and verdictBull ? colBullSoft : showBackground and verdictBear ? colBearSoft : na)

// ---------------------- DASHBOARD TABLE ----------------------
var table dash = table.new(position.top_right, 2, 6, bgcolor=colPanelBg, border_width=1, border_color=color.new(#363A45, 0))

voteLabel(v) =>
    v == 1 ? "Bullish" : v == -1 ? "Bearish" : "Neutral"

voteColor(v) =>
    v == 1 ? colBull : v == -1 ? colBear : colNeutral

if showDashboard and barstate.islast
    table.cell(dash, 0, 0, "CLAUDE TRADING INDICATOR", text_color=colPanelText, bgcolor=colPanelBg, text_size=size.small)
    table.cell(dash, 1, 0, str.tostring(bullCount) + "B / " + str.tostring(bearCount) + "S / " + str.tostring(enabledCount), text_color=colPanelText, bgcolor=colPanelBg, text_size=size.small)

    table.cell(dash, 0, 1, "Trend", text_color=colPanelText, text_size=size.small)
    table.cell(dash, 1, 1, useTrend ? voteLabel(trendVote) : "Off", text_color=useTrend ? voteColor(trendVote) : colNeutral, text_size=size.small)

    table.cell(dash, 0, 2, "Momentum", text_color=colPanelText, text_size=size.small)
    table.cell(dash, 1, 2, useMomentum ? voteLabel(momentumVote) : "Off", text_color=useMomentum ? voteColor(momentumVote) : colNeutral, text_size=size.small)

    table.cell(dash, 0, 3, "Volatility", text_color=colPanelText, text_size=size.small)
    table.cell(dash, 1, 3, useVolatility ? voteLabel(volatilityVote) : "Off", text_color=useVolatility ? voteColor(volatilityVote) : colNeutral, text_size=size.small)

    table.cell(dash, 0, 4, "Volume", text_color=colPanelText, text_size=size.small)
    table.cell(dash, 1, 4, useVolume ? voteLabel(volumeVote) : "Off", text_color=useVolume ? voteColor(volumeVote) : colNeutral, text_size=size.small)

    verdictText = verdictBull ? "BUY ZONE" : verdictBear ? "SELL ZONE" : "No Consensus"
    verdictCol  = verdictBull ? colBull : verdictBear ? colBear : colNeutral
    table.cell(dash, 0, 5, "Verdict", text_color=colPanelText, text_size=size.small)
    table.cell(dash, 1, 5, verdictText, text_color=verdictCol, text_size=size.small)

// ============================ ALERTS ============================
alertcondition(buySignal, title="Claude Trading Indicator - BUY", message="Claude Trading Indicator: BUY signal on {{ticker}} ({{interval}})")
alertcondition(sellSignal, title="Claude Trading Indicator - SELL", message="Claude Trading Indicator: SELL signal on {{ticker}} ({{interval}})")
